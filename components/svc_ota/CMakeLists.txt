# svc_ota — service layer.
#
# ota_policy.c: admission POLICY, pure, host-tested (Phase 0).
# ota_manifest_parse.c: canonical-bytes encoding + the full<->policy
# projection, pure, host-tested (Phase 2 Tier 0/4).
# ota_manifest_json.c: the strict manifest JSON extractor, pure (Tier 4) --
# attacker-controlled input, so every ambiguity is a reject; see its own
# header comment.
# ota_trust_anchors.c: table lookup of the two compiled-in ECDSA P-256
# trust anchors by key_id, pure (Tier 4) -- the actual mbedTLS verify is
# target-only (ota_verify.c, landing alongside the rest of the transport
# half).
# ota_fsm.c: the runtime OTA state machine on core_fsm_t (Tier 4) --
# EVENTS_AND_STATE.md §6.4 lists `ota` among the FSMs that must be
# table-driven. Needs core_fsm; still no ESP-IDF/FreeRTOS.
#
# All four are pure with minimal REQUIRES: policy and the verify-adjacent
# modules take injected callbacks/tables rather than calling mbedTLS
# directly, the canonical encoding and JSON parser are bytes-in/bytes-out
# arithmetic, and the FSM's side effects are injected via a vtable (same
# pattern as net_wifi_fsm.c/svc_supervisor.c). That is what lets all of
# them be tested deterministically with no key material and no ESP-IDF at
# all.
#
# Target-only glue (Tier 4d-f):
# ota_verify.c: real ota_verifier_t callbacks -- mbedTLS ECDSA-P256 manifest
# verify, an independent esp_image_verify() image-signature re-check.
# svc_ota.c: construction (boot-time PENDING_VERIFY detection per
# LIFECYCLE.md §1.4), the tick/gate-relay entry points, and the FSM actions
# that only log/publish.
# svc_ota_download.c: the transient "ota" task -- net_http_stream_*() (never
# esp_https_ota, see its own header comment) feeding esp_ota_write() and a
# running mbedtls_sha256 in the same loop.
# svc_ota_apply.c: the three FSM actions that end in
# esp_ota_set_boot_partition()/esp_ota_mark_app_valid_cancel_rollback()/
# esp_ota_mark_app_invalid_rollback_and_reboot().
#
# ota_trust_anchor_dev.c (Tier 5): GENERATED by tools/gen_trust_anchor.py
# from tools/keys/dev_manifest_signing.pem's public half -- see that
# script's own docstring for why it's committed (embeds only a public key)
# despite depending on a gitignored private key.

idf_component_register(
    SRCS        "ota_policy.c" "ota_manifest_parse.c" "ota_manifest_json.c"
                "ota_manifest_codec.c" "ota_trust_anchors.c" "ota_fsm.c"
                "ota_verify.c" "svc_ota.c" "svc_ota_download.c" "svc_ota_apply.c"
                "ota_trust_anchor_dev.c"
    INCLUDE_DIRS "include"
    REQUIRES     "core_types" "core_eventbus" "core_fsm" "net_http" "esp_partition"
    PRIV_REQUIRES "app_update" "mbedtls" "bootloader_support" "esp_timer"
)
